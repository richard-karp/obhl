```ts
/** Small seeded PRNG; every schedule phase draws from its own seed. */
export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

```ts
import { leagueDateKey } from "@/lib/format";

/** Whether a game's league-zone date is before `today` (a `leagueToday()` key). */
export function isPastGame(
  g: { scheduled_at: string | null },
  today: string,
): boolean {
  return !!g.scheduled_at && leagueDateKey(g.scheduled_at) < today;
}

/** Past games never finalized: still `scheduled` or `in_progress`. */
export function openPastGames<
  G extends { status: string; scheduled_at: string | null },
>(games: G[], today: string): G[] {
  return games.filter(
    (g) =>
      (g.status === "scheduled" || g.status === "in_progress") &&
      isPastGame(g, today),
  );
}
```

```tsx
/**
 * Past games nobody finalized. A night that fails to close shows up nowhere
 * else. Admin client: manager-gated, so it must not depend on the season being
 * public (see `getSchedule`).
 */
async function OpenGamesCard({
  seasonId,
  leagueSlug,
}: {
  seasonId: string | null;
  leagueSlug: string;
}) {
  if (!seasonId) return null;
  const games = openPastGames(
    await getSchedule(seasonId, { client: createAdminClient() }),
    leagueToday(),
  );
  if (games.length === 0) return null;

  return (
    <section aria-label="Games still open">
      <Card>
        <CardHeader>
          <CardTitle className="text-base">Games still open</CardTitle>
        </CardHeader>
        <CardContent className="space-y-3">
          <p className="text-muted-foreground text-sm">
            These games are in the past and were never finalized.
          </p>
          <div className="divide-y rounded-lg border">
            {games.map((g) => (
              <div
                key={g.id}
                className="flex items-center justify-between gap-3 px-3 py-2"
              >
                <span className="flex items-center gap-2 text-sm">

      </Card>
    </section>
  );
}
```

## Appendix A: `comment-only.cjs`

Save to the scratchpad, verbatim:

```js
// Proves a diff changes only comments: parse each changed .ts/.tsx/.mts/.mjs file at BASE and in
// the working tree, print with comments removed, collapse whitespace, compare.
// Usage: node comment-only.cjs <base-ref>   (exit 0 = comment-only)
// Known gap: collapsing whitespace hides a whitespace-only edit inside a string literal.
// Pair with: git diff -U0 <base> -- '*.ts' '*.tsx' '*.mjs' | grep -E '^-.*(eslint-disable|@ts-(expect-error|ignore|nocheck)|/// <reference)'
// which must print nothing (removing a directive is comment-only to the AST but not to lint/tsc).
const ts=require(require.resolve("typescript",{paths:[process.cwd()]})),fs=require("fs"),{execFileSync}=require("child_process");
const base=process.argv[2], git=(...a)=>execFileSync("git",a,{encoding:"utf8",maxBuffer:1<<28});
const kind=f=>f.endsWith(".tsx")?ts.ScriptKind.TSX:/\.m?js$/.test(f)?ts.ScriptKind.JS:ts.ScriptKind.TS;
const norm=(f,t)=>ts.createPrinter({removeComments:true})
  .printFile(ts.createSourceFile(f,t,ts.ScriptTarget.Latest,true,kind(f))).replace(/\s+/g," ");
const files=git("diff","--name-only",base,"--","*.ts","*.tsx","*.mts","*.mjs").split("\n").filter(Boolean);
let bad=0;
for(const f of files){let b=null;try{b=git("show",`${base}:${f}`)}catch{}
  const a=fs.existsSync(f)?fs.readFileSync(f,"utf8"):null;
  if(b===null||a===null){console.log("ADDED/DELETED:",f);bad=1;continue}
  if(norm(f,b)!==norm(f,a)){console.log("CODE CHANGED:",f);bad=1}}
console.log(files.length+" files checked");process.exit(bad);
```

```js
// Counts comment-only, code and blank lines in the .ts/.tsx files under each path given.
// Usage, from the repo root: node count-lines.cjs src/lib/schedule e2e next.config.ts
// Uses the parser, not a bare scanner: a scanner misreads regex literals. JSDoc parsing is off,
// or /** */ blocks become AST nodes and count as code. A line with code and a comment is code.
const ts = require(require.resolve("typescript", { paths: [process.cwd()] }));
const fs = require("fs");
const path = require("path");

function classify(file) {
  const text = fs.readFileSync(file, "utf8");
  const sf = ts.createSourceFile(
    file,
    text,
    { languageVersion: ts.ScriptTarget.Latest, jsDocParsingMode: ts.JSDocParsingMode.ParseNone },
    true,
    file.endsWith(".tsx") ? ts.ScriptKind.TSX : ts.ScriptKind.TS,
  );
  const starts = [0];
  for (let i = 0; i < text.length; i++) if (text[i] === "\n") starts.push(i + 1);
  const lineOf = (p) => {
    let lo = 0, hi = starts.length - 1;
    while (lo < hi) {
      const m = (lo + hi + 1) >> 1;
      if (starts[m] <= p) lo = m;
      else hi = m - 1;
    }
    return lo;
  };
  const status = new Array(starts.length).fill("blank");
  const mark = (a, b, kind) => {
    if (b <= a) return;
    for (let l = lineOf(a); l <= lineOf(b - 1); l++)
      if (kind === "code" || status[l] === "blank") status[l] = kind;
  };
  (function visit(node) {
    const kids = node.getChildren(sf);
    if (kids.length) return kids.forEach(visit);
    const start = node.getStart(sf);
    for (const r of ts.getLeadingCommentRanges(text, node.pos) || [])
      if (r.pos < start) mark(r.pos, Math.min(r.end, start), "comment");
    mark(start, node.end, "code");
  })(sf);
  const lines = text.endsWith("\n") ? starts.length - 1 : starts.length;
  const out = { comment: 0, code: 0, blank: 0 };
  for (let l = 0; l < lines; l++) out[status[l]]++;
  return out;
}

function files(p) {
  if (!fs.existsSync(p)) return [];
  if (fs.statSync(p).isFile()) return /\.tsx?$/.test(p) ? [p] : [];
  return fs
    .readdirSync(p, { withFileTypes: true })
    .filter((e) => e.name !== "node_modules" && e.name !== ".next")
    .flatMap((e) => files(path.join(p, e.name)));
}

for (const p of process.argv.slice(2)) {
  const t = { files: 0, comment: 0, code: 0, blank: 0 };
  for (const f of files(p)) {
    const r = classify(f);
    t.files++;
    t.comment += r.comment;
    t.code += r.code;
    t.blank += r.blank;
  }
  console.log(`${p}\tfiles ${t.files}\tcomment ${t.comment}\tcode ${t.code}\tblank ${t.blank}`);
}
```
